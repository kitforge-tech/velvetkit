class InvitationsController < ApplicationController
  def new
    @invitation = Invitation.new
  end

  def create
    @invitation = Invitation.new(invitation_params)

    if @invitation.save
      redirect_to team_path(@invitation.team), notice: "Invite sent to #{@invitation.email}"
    else
      render :new, status: :unprocessable_entity
    end
  end

  private

  def invitation_params
    params.expect(invitation: [ :name, :email, :role, :team_id, :starts_on, :note, :notify, :terms ])
  end
end
